This is the sample curl to create stack

curl --location 'https://api.contentstack.io/v3/stacks' \
--header 'authtoken: ble63aed49b6877' \
--header 'organization_uid: your_organization_uid' \
--header 'Content-Type: application/json' \
--data '{
  "stack": {
    "name": "My New Stack",
    "description": "My new test stack",
    "master_locale": "en-us"
  }
}'
